  // PHASE 1: Update hero section with live data
  function updateHeroSection() {
    if (!window.heroController) return;

    // Calculate P&L today
    const today = new Date().toDateString();
    const tradesData = state._ordersCache || [];
    let pnlToday = 0;
    let lastTrade = null;

    if (Array.isArray(tradesData)) {
      tradesData.forEach((trade) => {
        const tradeDate = new Date(trade.timestamp || trade.time).toDateString();
        if (tradeDate === today && trade.realizedProfit) {
          pnlToday += Number(trade.realizedProfit) || 0;
          if (!lastTrade || new Date(trade.timestamp || trade.time) > new Date(lastTrade.timestamp || lastTrade.time)) {
            lastTrade = trade;
          }
        }
      });
    }

    // Calculate P&L percentage
    const estimatedDailyCapital = 10000;
    const pnlPercent = estimatedDailyCapital > 0 ? (pnlToday / estimatedDailyCapital) * 100 : 0;

    // Update P&L display
    window.heroController.updatePnL(pnlToday, pnlPercent);

    // Update last action
    if (lastTrade) {
      const side = lastTrade.side || (pnlToday >= 0 ? 'BUY' : 'SELL');
      const symbol = lastTrade.symbol || state.symbol;
      const price = lastTrade.price || '--';
      const timeAgo = getTimeAgoText(new Date(lastTrade.timestamp || lastTrade.time));
      window.heroController.updateLastAction(`${side} ${symbol} @ ${price} • ${timeAgo}`);
    } else {
      window.heroController.updateLastAction('No operations yet');
    }

    // Update confidence
    const confidence = Math.min(100, Math.max(0, state.signal?.strength || 0) * 100);
    window.heroController.updateConfidence(confidence);
  }

  function getTimeAgoText(date) {
    const now = new Date();
    const diffMs = now - date;
    const diffMins = Math.floor(diffMs / 60000);
    const diffHours = Math.floor(diffMins / 60);
    const diffDays = Math.floor(diffHours / 24);

    if (diffMins < 1) return 'just now';
    if (diffMins < 60) return `${diffMins}m ago`;
    if (diffHours < 24) return `${diffHours}h ago`;
    if (diffDays < 30) return `${diffDays}d ago`;
    return date.toLocaleDateString();
  }
